	#
	#   修法：移除科技要求，使默认 PM 与建筑同步可用。
	#

	#   「组内无人标 default 时，第一个 PM 即默认」，显式标记完全冗余。
	#   删除后行为不变，回归香草写法 —— 香草 365 个 PM 中仅 9 个标了
	#   is_default，且都用于「覆盖非第一个 PM」的场景。
	# is_default = yes

	# 原科技要求（已按第十轮修复移除）：
