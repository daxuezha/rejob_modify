# 开启 / 重置「失业人群管理中心」日志（可重复点击）
# ----------------------------------------------------------------------------
# 行为：
#   · 日志尚未开启 → 直接添加日志条目（按 immediate 走一遍默认初始化）
#   · 日志已开启   → 执行 rejob_reset_all：清空调节值与两个开关，
#                    重新初始化为默认状态（调节值 0、两个开关关闭），
#                    并刷新动态修正，等价于「删掉旧日志、生成一条全新的」。
#
# 注：Victoria 3 的脚本系统没有「删除日志条目」的效果
#     （invalidate/complete/fail_journal_entry 均为控制台命令，非脚本效果），
#     因此这里用「就地重置为默认状态」实现同一效果：对玩家而言，
#     结果与「先删除、再新建」完全一致（所有值回到默认）。
# ============================================================================
rejob_open_management_decision = {
	is_shown = {
		is_player = yes
		rejob_has_any_center = yes
	}

	possible = {
		always = yes
	}

	when_taken = {
		if = {
			limit = { NOT = { has_journal_entry = je_rejob_management } }
			add_journal_entry = {
				type = je_rejob_management
			}
		}
		else = {
			rejob_reset_all = yes
		}
	}
}
