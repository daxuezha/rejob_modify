# A. 死亡率调节按钮
# ---------------------------------------------------------------------------
# 单位：rejob_offset 使用 state_mortality_mult 的「原始修正单位」，
#       按钮写 0.1 即代表 +10%（详见文件顶部单位约定）。
# 下限保护：offset 不低于 -1000（即 -100000%），防止误操作后变量失控。
# 合计下限：由脚本值的 min = 0 保证「合计不为负」（见 script_values）。
# 说明：负调节值会被 rejob_uc_offset_effective 的
#       min = -生产方式基准值 截断，最多只能抵消掉 PM 自带的死亡率，
#       因此合计最低为 0，不会出现负死亡率。
# ===========================================================================

# ===========================================================================
# B. 两个「生效开关」按钮（对应需求 6）
# ---------------------------------------------------------------------------
# 决定调节值作用于哪座建筑：
#   两个都高亮 → 同时作用于「再就业中心」与「集中营」
#   只高亮一个 → 只作用于那一座
#   两个都灭   → 都不作用（调节值保留，但不产生任何修正）
#
# 开关只控制「修改动作是否生效」：
#   点击开关不会改动 rejob_offset；
#   对被关闭的建筑是撤销其动态修正，对被开启的建筑是重新施加，
#   调节值本身保持原样 —— 符合需求中「修改动作的生效」而非「已修改数值变化」。
#
# possible 要求对应建筑存在，否则置灰（需求 7）。
# ===========================================================================

rejob_toggle_apply_uc = {
	name = "rejob_toggle_apply_uc"
	desc = "rejob_toggle_apply_uc_desc"
	possible = {
		rejob_has_unemployment_center = yes
	}

	# 开关为 1 时按钮高亮（OGAS 同款视觉反馈）
	selected = {
		root.var:rejob_apply_uc = 1
	}

	effect = {
		if = {
			limit = { root.var:rejob_apply_uc = 1 }
			set_variable = { name = rejob_apply_uc  value = 0 }
		}
		else = {
			set_variable = { name = rejob_apply_uc  value = 1 }
		}
		rejob_refresh_mortality = yes
	}
}

# --- 原 mod 决议：066 号指令（不可接触者） ---
rejob_btn_decision_066 = {
	name = "rejob_btn_decision_066"
	desc = "rejob_btn_decision_066_desc"
	possible = {
		owner = {
			is_player = yes
			NOT = { has_law = law_type:law_racial_segregation }
			NOT = { has_law = law_type:law_cultural_exclusion }
			NOT = { has_law = law_type:law_multicultural }
			NOT = { has_law = law_type:law_subjecthood }
		}
		# 用 hidden_trigger 包裹，避免引擎为 any_scope_culture 里的
		# cultural_acceptance_base 生成「依赖作用域」的本地化提示而刷 warning.log。
		hidden_trigger = {
			any_scope_culture = {
				ROOT = {
					cultural_acceptance_base = { target = prev  value < acceptance_status_4 }
				}
			}
		}
	}

	effect = {
		rejob_run_decision_066 = yes
	}
}

# --- 原 mod 决议：067 号指令（冤假错案） ---
rejob_btn_decision_067 = {
	name = "rejob_btn_decision_067"
	desc = "rejob_btn_decision_067_desc"
	possible = {
		# 同样用 hidden_trigger 包裹，避免 state_cultural_acceptance
		# 在 any/every 触发器内生成作用域依赖的本地化提示。
		#
		# ★ 2026-10-14 修复：原写法在【触发器上下文】里用了 `limit = { ... }`
		#   报错（error.3.log:2721 / debug.1.log:977）：
		#     Error: "Unknown trigger type: limit, near line: 233"
		#       in file: "common/scripted_buttons/rejob_mortality_buttons.txt" near line: 242
		#   `limit` 是【效果上下文】的关键字，不能出现在 `possible` 这种触发器里。
		#   香草实证：把 `limit` 写在 `any_scope_pop` 块内，全库 0 处；
		#   正确写法是把条件【内联】直接写进 any_scope_pop 块内
		#   （香草 00_decree.txt:170、01_character_ideologies.txt:4727 均如此）。
		hidden_trigger = {
			any_scope_pop = {
				state = {
					state_cultural_acceptance = {
						target = PREV.culture
						value > acceptance_status_3
					}
				}
				is_pop_type = delicious
			}
		}
	}

	effect = {
		rejob_run_decision_067 = yes
	}
}

# ★ 2026-10-18【v2 方案】D. 石油 / 橡胶产出比例调节（0 = 纯橡胶，100 = 纯石油）
# ---------------------------------------------------------------------------
# 变量：rejob_oil_ratio，范围 [0, 100]，默认 100（= 纯石油，与改动前一致）
# 单位：这是「百分比数值」，不是 mortality 那种修正单位 ——
#       100 表示「100% 石油」，0 表示「100% 橡胶」。
# 上下限：达到 0 或 100 后对应方向按钮置灰（见 rejob_oil_ratio_can_up/down），
#         且 effect 内再做一次守卫（双保险，变量永不越界）。
# 说明：ratio 是「石油占比」；实际产出的石油/橡胶比例由 _mult 修正实现，
#       见 static_modifiers/rejob_mortality_modifiers.txt 的 4 个 _mult 容器。
# ===========================================================================

# ★ 2026-10-18【常驻化 · 需求 2】E. 产出比例的两个「编辑锁」
# ---------------------------------------------------------------------------
# ★ 语义变更（重要）：
#   旧：开关 = 「是否施加比例修正」  ⇒ 关闭后产出回到 2000+2000（50/50）
#   新：开关 = 「是否允许修改比例」  ⇒ 关闭后【修正仍然生效】，只是不能改数值
#
#   为什么改：PM 声明了「2000 油 + 2000 橡胶」这一固定约束，
#   若修正可被关闭，则产出恒为 50/50，玩家无法回到「纯石油」，
#   与该生产方式应有的默认语义冲突（需求方 2026-10-18 反馈）。
#
#   现在的行为：
#     · 比例修正【永远挂着】，ratio 默认 100 ⇒ 纯石油（默认即正确）
#     · 开启开关 ⇒ 可以点击 ± 按钮调整 ratio
#     · 关闭开关 ⇒ 按钮置灰，但修正照常生效，产出不变
# ===========================================================================

rejob_toggle_apply_oil_uc = {
	name = "rejob_toggle_apply_oil_uc"
	desc = "rejob_toggle_apply_oil_uc_desc"
	possible = {
		rejob_has_unemployment_center = yes
		has_variable = rejob_apply_oil_uc
	}

	selected = {
		has_variable = rejob_apply_oil_uc
		root.var:rejob_apply_oil_uc = 1
	}

	effect = {
		if = {
			limit = { has_variable = rejob_apply_oil_uc  root.var:rejob_apply_oil_uc = 1 }
			set_variable = { name = rejob_apply_oil_uc  value = 0 }
		}
		else = {
			set_variable = { name = rejob_apply_oil_uc  value = 1 }
		}
		rejob_refresh_oil_balance = yes
	}
}
