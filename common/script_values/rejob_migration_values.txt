# 估算「非首都州失业人口」总量。
# 用途：迁移决议 / 日志按钮的可用性门槛。
# 说明：该值为近似估算（州总人口 × 失业率之和），不代表精确的合格人数，
#       实际可迁移人数可能更少（事件内还会排除 delicious 人口）。
rejob_noncapital_unemployed_population_estimate = {
	value = 0
	every_scope_state = {
		limit = {
			is_capital = no
		}
		add = {
			value = state_population
			multiply = state_unemployment_rate
		}
	}
}
