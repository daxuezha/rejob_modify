		name = rejob_migration.1.a
		trigger = {
			rejob_noncapital_unemployed_population_estimate >= 1000000
		}
		custom_tooltip = rejob_migration.1.a_tt
		hidden_effect = {
			set_variable = { name = rejob_migration_remaining value = 1000000 }
			set_variable = { name = rejob_migration_moved value = 0 }
			set_variable = { name = rejob_migration_chunk value = 0 }
			every_scope_state = {
				limit = { is_capital = no }
				every_scope_pop = {
					limit = {
						is_employed = no
						NOT = { is_pop_type = delicious }
					}
					if = {
						limit = { root.var:rejob_migration_remaining > 0 }
						save_scope_as = rejob_migration_current_pop
						root = {
							set_variable = {
								name = rejob_migration_chunk
								value = {
									value = scope:rejob_migration_current_pop.total_size
									max = var:rejob_migration_remaining
									floor = yes
								}
							}
						}
						if = {
							limit = { root.var:rejob_migration_chunk > 0 }
							move_partial_pop = {
								state = root.capital
								population = root.var:rejob_migration_chunk
							}
							root = {
								change_variable = {
									name = rejob_migration_remaining
									add = {
										value = var:rejob_migration_chunk
										multiply = -1
									}
								}
								change_variable = {
									name = rejob_migration_moved
									add = var:rejob_migration_chunk
								}
							}
						}
					}
				}
			}
			add_treasury = {
				value = var:rejob_migration_moved
				multiply = -1
			}
			set_variable = { name = rejob_migration_remaining value = 0 }
			set_variable = { name = rejob_migration_moved value = 0 }
			set_variable = { name = rejob_migration_chunk value = 0 }
		}
